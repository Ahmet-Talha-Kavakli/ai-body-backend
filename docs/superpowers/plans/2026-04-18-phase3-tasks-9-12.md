```typescript
import { getDatabase } from './sqlite'
import type { CoachQuestion, CoachResponse } from '../types/nutrition-coach'

export async function createCoachTables(): Promise<void> {
  const db = await getDatabase()

  // Create coach_questions table
  await db.runAsync(
    `
    CREATE TABLE IF NOT EXISTS coach_questions (
      id TEXT PRIMARY KEY,
      userId TEXT NOT NULL,
      question TEXT NOT NULL,
      voiceUrl TEXT,
      timestamp TEXT NOT NULL,
      createdAt INTEGER NOT NULL
    )
    `
  )

  // Create coach_responses table
  await db.runAsync(
    `
    CREATE TABLE IF NOT EXISTS coach_responses (
      id TEXT PRIMARY KEY,
      questionId TEXT NOT NULL,
      userId TEXT NOT NULL,
      response TEXT NOT NULL,
      source TEXT NOT NULL,
      timestamp TEXT NOT NULL,
      createdAt INTEGER NOT NULL,
      FOREIGN KEY (questionId) REFERENCES coach_questions(id)
    )
    `
  )
}

export async function saveQuestion(question: CoachQuestion): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(
    `INSERT INTO coach_questions (id, userId, question, voiceUrl, timestamp, createdAt)
     VALUES (?, ?, ?, ?, ?, ?)`,
    [
      question.id,
      question.userId,
      question.question,
      question.voiceUrl || null,
      question.timestamp,
      Date.now(),
    ]
  )
}

export async function saveResponse(response: CoachResponse): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(
    `INSERT INTO coach_responses (id, questionId, userId, response, source, timestamp, createdAt)
     VALUES (?, ?, ?, ?, ?, ?, ?)`,
    [
      response.id,
      response.questionId,
      response.userId,
      response.response,
      response.source,
      response.timestamp,
      Date.now(),
    ]
  )
}

export async function getConversationHistory(
  userId: string
): Promise<
  Array<{ type: 'question' | 'response'; id: string; content: string; timestamp: string }>
> {
  const db = await getDatabase()

  // Get all questions
  const questions = await db.getAllAsync(
    `SELECT id, question as content, timestamp FROM coach_questions WHERE userId = ? ORDER BY timestamp ASC`,
    [userId]
  )

  // Get all responses
  const responses = await db.getAllAsync(
    `SELECT id, response as content, timestamp FROM coach_responses WHERE userId = ? ORDER BY timestamp ASC`,
    [userId]
  )

  // Merge and sort by timestamp
  const combined = [
    ...questions.map((q: any) => ({ type: 'question' as const, ...q })),
    ...responses.map((r: any) => ({ type: 'response' as const, ...r })),
  ]

  return combined.sort((a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime())
}
```

```typescript
import React, { useRef, useEffect } from 'react'
import { View, ScrollView, Text } from 'react-native'

interface Message {
  id: string
  type: 'user' | 'coach'
  text: string
  timestamp: string
}

interface CoachChatProps {
  messages: Message[]
  isLoading: boolean
}

export function CoachChat({ messages, isLoading }: CoachChatProps) {
  const scrollViewRef = useRef<ScrollView>(null)

  // Auto-scroll to bottom when new messages arrive
  useEffect(() => {
    if (scrollViewRef.current) {
      setTimeout(() => {
        scrollViewRef.current?.scrollToEnd({ animated: true })
      }, 100)
    }
  }, [messages, isLoading])

  return (
    <ScrollView
      ref={scrollViewRef}
      testID="chat-container"
      className="flex-1 bg-white px-4 py-2"
      showsVerticalScrollIndicator={true}
    >
      {messages.map((message) => (
        <View
          key={message.id}
          className={`mb-3 flex ${message.type === 'user' ? 'flex-row-reverse' : 'flex-row'}`}
        >
          <View
            className={`max-w-xs rounded-lg px-3 py-2 ${
              message.type === 'user'
                ? 'bg-blue-500'
                : 'bg-gray-200'
            }`}
          >
            <Text
              className={`text-base ${
                message.type === 'user' ? 'text-white' : 'text-gray-800'
              }`}
            >
              {message.text}
            </Text>
          </View>
        </View>
      ))}

      {isLoading && (
        <View testID="typing-indicator" className="mb-3 flex flex-row">
          <View className="bg-gray-200 rounded-lg px-3 py-2">
            <Text className="text-gray-500">Coach is typing...</Text>
          </View>
        </View>
      )}
    </ScrollView>
  )
}
```

```typescript
import React, { useState, useCallback } from 'react'
import { View, TextInput, TouchableOpacity, Text } from 'react-native'
import { CoachChat } from '../../components/nutrition/CoachChat'
import { useCoachContext } from '../../hooks/useCoachContext'
import { getPersonalizedAdvice } from '../../services/claudeService'
import { saveQuestion, saveResponse } from '../../db/nutritionCoach'
import { useAuthStore } from '../../store/authStore'

interface Message {
  id: string
  type: 'user' | 'coach'
  text: string
  timestamp: string
}

export function NutritionCoachScreen() {
  const [messages, setMessages] = useState<Message[]>([])
  const [inputText, setInputText] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  const { buildCoachContext } = useCoachContext()
  const authStore = useAuthStore()

  // Simple UUID generator
  const generateId = () => Math.random().toString(36).substr(2, 9)

  const handleSendMessage = useCallback(
    async (text: string) => {
      if (!text.trim() || !authStore.userEmail) return

      try {
        setIsLoading(true)

        // 1. Save user question to DB
        const questionId = generateId()
        const userMessage: Message = {
          id: questionId,
          type: 'user',
          text,
          timestamp: new Date().toISOString(),
        }

        await saveQuestion({
          id: questionId,
          userId: authStore.userEmail,
          question: text,
          timestamp: new Date().toISOString(),
        })

        // Add to chat
        setMessages((prev) => [...prev, userMessage])
        setInputText('')

        // 2. Build coach context
        const context = await buildCoachContext(authStore.userEmail)

        // 3. Get Claude response
        const response = await getPersonalizedAdvice(text, context)

        // 4. Save response to DB
        const responseId = generateId()
        await saveResponse({
          id: responseId,
```

```typescript
      id INTEGER PRIMARY KEY AUTOINCREMENT,

    `

export async function addWaterLog(date: string, ml: number, userId?: string): Promise<void> {
  const db = await getDatabase()
  const goalMl = 2000 // default 2L

  // Try to update existing entry for the date, insert if not exists
  const existing = await db.getFirstAsync(
    `SELECT id, totalMl FROM water_intake WHERE date = ? LIMIT 1`,
    [date]
  )

  if (existing) {
    // Update existing
    await db.runAsync(`UPDATE water_intake SET totalMl = totalMl + ? WHERE date = ?`, [ml, date])
  } else {
    // Insert new
    await db.runAsync(
      `INSERT INTO water_intake (userId, date, totalMl, goalMl, timestamp, createdAt)
       VALUES (?, ?, ?, ?, ?, ?)`,
      [userId || 'default', date, ml, goalMl, new Date().toISOString(), Date.now()]
    )
  }
}

export async function getWaterForDate(date: string): Promise<{ totalMl: number; goalMl: number }> {
  const db = await getDatabase()

  const result = await db.getFirstAsync(
    `SELECT totalMl, goalMl FROM water_intake WHERE date = ? LIMIT 1`,
    [date]
  )

  return result || { totalMl: 0, goalMl: 2000 }
}

export async function getWaterStats(startDate: string, endDate: string): Promise<WaterDaily[]> {
  const db = await getDatabase()

  const results = await db.getAllAsync(
    `SELECT date, totalMl FROM water_intake WHERE date >= ? AND date <= ? ORDER BY date ASC`,
    [startDate, endDate]
  )

  return results || []
}
```

```typescript
import { getDatabase } from './sqlite'
import type { NutritionStreak } from '../types/nutrition-streak'

export async function createStreakTable(): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(
    `
    CREATE TABLE IF NOT EXISTS nutrition_streaks (
      userId TEXT PRIMARY KEY,
      currentStreak INTEGER NOT NULL DEFAULT 0,
      longestStreak INTEGER NOT NULL DEFAULT 0,
      lastLogDate TEXT,
      createdAt INTEGER NOT NULL,
      updatedAt INTEGER NOT NULL
    )
    `
  )
}

export async function logMealForToday(userId: string): Promise<void> {
  const db = await getDatabase()
  const today = new Date().toISOString().split('T')[0]

  // Get current streak
  const existing = await db.getFirstAsync(
    `SELECT currentStreak, longestStreak, lastLogDate FROM nutrition_streaks WHERE userId = ?`,
    [userId]
  )

  if (!existing) {
    // First meal logged - start streak at 1
    await db.runAsync(
      `INSERT INTO nutrition_streaks (userId, currentStreak, longestStreak, lastLogDate, createdAt, updatedAt)
       VALUES (?, ?, ?, ?, ?, ?)`,
      [userId, 1, 1, today, Date.now(), Date.now()]
    )
  } else {
    const lastDate = existing.lastLogDate
    const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0]

    // Check if streak continues (meal logged today or yesterday)
    let newStreak = existing.currentStreak
    if (lastDate !== today && lastDate !== yesterday) {
      newStreak = 1 // streak broken, restart
    } else if (lastDate !== today) {
      newStreak = existing.currentStreak + 1 // continue streak
    }

    const newLongest = Math.max(newStreak, existing.longestStreak)

    await db.runAsync(
      `UPDATE nutrition_streaks SET currentStreak = ?, longestStreak = ?, lastLogDate = ?, updatedAt = ? WHERE userId = ?`,
      [newStreak, newLongest, today, Date.now(), userId]
    )
  }
}

export async function getStreak(userId: string): Promise<NutritionStreak> {
  const db = await getDatabase()

  const result = await db.getFirstAsync(
    `SELECT userId, currentStreak, longestStreak, lastLogDate FROM nutrition_streaks WHERE userId = ?`,
    [userId]
  )

  return result || { userId, currentStreak: 0, longestStreak: 0, lastLogDate: '' }
}

export async function checkStreakBreak(userId: string): Promise<void> {
  const db = await getDatabase()

  const existing = await db.getFirstAsync(
    `SELECT lastLogDate, longestStreak FROM nutrition_streaks WHERE userId = ?`,
    [userId]
  )

  if (existing && existing.lastLogDate) {
    const lastDate = new Date(existing.lastLogDate)
    const today = new Date()
    const daysDiff = Math.floor((today.getTime() - lastDate.getTime()) / 86400000)

    // If more than 1 day has passed, streak is broken
    if (daysDiff > 1) {
      await db.runAsync(
        `UPDATE nutrition_streaks SET currentStreak = 0, updatedAt = ? WHERE userId = ?`,
        [Date.now(), userId]
      )
    }
  }
}
```

```typescript
interface StreakCalendarProps {

export function StreakCalendar({ currentStreak, loggedDates }: StreakCalendarProps) {
  // Generate 49 dates (7 weeks back from today)
  const today = new Date()
  const dates: Date[] = []

  for (let i = 48; i >= 0; i--) {
    const date = new Date(today)
    date.setDate(date.getDate() - i)
    dates.push(date)
  }

  const dateStrings = new Set(loggedDates)

  const getCellColor = (date: Date): string => {
    const dateStr = date.toISOString().split('T')[0]
    return dateStrings.has(dateStr) ? '#22c55e' : '#d1d5db' // green or gray
  }

  const getDayLabel = (date: Date): string => {
    return date.getDate().toString()
  }

  return (
    <View className="bg-white rounded-lg p-4 gap-4">
      {/* Streak header */}
      <View className="items-center">
        <Text className="text-2xl font-bold text-gray-800">
          {currentStreak} days 🔥
        </Text>
      </View>

      {/* Calendar grid */}
      <View>
        {/* Header: days of week */}
        <View className="flex-row justify-between mb-2">
          {['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].map((day) => (
            <Text key={day} className="w-12 text-center text-xs font-semibold text-gray-600">
```

```typescript
import { getDatabase } from './sqlite'
import type {
  NutritionSyncItem,
  MealData,
  PhotoData,
  GoalData,
  SyncType,
} from '../types/nutrition-sync'

export async function createSyncQueueTable(): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(
    `
    CREATE TABLE IF NOT EXISTS nutrition_sync_queue (
      id TEXT PRIMARY KEY,
      userId TEXT NOT NULL,
      type TEXT NOT NULL,
      payload TEXT NOT NULL,
      status TEXT NOT NULL,
      attempts INTEGER NOT NULL DEFAULT 0,
      error TEXT,
      createdAt TEXT NOT NULL,
      lastAttemptAt TEXT,
      updatedAt INTEGER NOT NULL
    )
    `
  )
}

export async function queueMealSync(meal: MealData): Promise<string> {
  const db = await getDatabase()
  const id = Math.random().toString(36).substr(2, 9)

  await db.runAsync(
    `INSERT INTO nutrition_sync_queue (id, userId, type, payload, status, attempts, createdAt, updatedAt)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      id,
      meal.userId,
      'meal',
      JSON.stringify(meal),
      'pending',
      0,
      new Date().toISOString(),
      Date.now(),
    ]
  )

  return id
}

  await db.runAsync(
    `INSERT INTO nutrition_sync_queue (id, userId, type, payload, status, attempts, createdAt, updatedAt)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      id,
      photo.userId,
      'photo',
      JSON.stringify(photo),
      'pending',
      0,
      new Date().toISOString(),
      Date.now(),
    ]

export async function queueGoalSync(goal: GoalData): Promise<string> {
  const db = await getDatabase()
  const id = Math.random().toString(36).substr(2, 9)

  await db.runAsync(
    `INSERT INTO nutrition_sync_queue (id, userId, type, payload, status, attempts, createdAt, updatedAt)
     VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
    [
      id,
      goal.userId,
      'goal',
      JSON.stringify(goal),
      'pending',
      0,
      new Date().toISOString(),

export async function getPendingItems(): Promise<NutritionSyncItem[]> {
  const db = await getDatabase()

  const items = await db.getAllAsync(
    `SELECT id, userId, type, payload, status, attempts, error, createdAt, lastAttemptAt
     FROM nutrition_sync_queue
     WHERE status = 'pending' OR status = 'failed'
     ORDER BY createdAt ASC`
  )

  return items || []
}

export async function markSynced(id: string): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(`UPDATE nutrition_sync_queue SET status = ?, updatedAt = ? WHERE id = ?`, [
    'synced',
    Date.now(),
    id,
  ])
}

export async function markFailed(id: string, error: string): Promise<void> {
  const db = await getDatabase()

  await db.runAsync(
    `UPDATE nutrition_sync_queue
     SET status = ?, error = ?, updatedAt = ?
     WHERE id = ?`,
    ['failed', error, Date.now(), id]
  )
}

export async function retryWithBackoff(id: string): Promise<number> {
  const db = await getDatabase()

  const item = await db.getFirstAsync(`SELECT attempts FROM nutrition_sync_queue WHERE id = ?`, [
    id,
  ])

  if (!item) {
    return 0
  }

  const attempts = item.attempts + 1
  const delayMs = Math.pow(2, attempts - 1) * 1000 // 1s, 2s, 4s, 8s, 16s

  await db.runAsync(
    `UPDATE nutrition_sync_queue
     SET attempts = ?, lastAttemptAt = ?, updatedAt = ?
     WHERE id = ?`,
    [attempts, new Date().toISOString(), Date.now(), id]
  )

  return delayMs
}
```

```typescript
import { useEffect, useRef } from 'react'
import { useOfflineDetection } from './useOfflineDetection'
import { useAuthStore } from '../store/authStore'
import { getAuthenticatedClient } from '../api/client'
import { getPendingItems, markSynced, markFailed, retryWithBackoff } from '../db/nutritionSync'

export function useNutritionSync() {
  const { isOnline } = useOfflineDetection()
  const syncInProgressRef = useRef(false)
  const retryTimeoutsRef = useRef<Map<string, NodeJS.Timeout>>(new Map())
  const authStore = useAuthStore()

  // Auto-sync when coming online
  useEffect(() => {
    if (isOnline && authStore.userEmail) {
      flushSyncQueue()
    }
  }, [isOnline, authStore.userEmail])

  async function flushSyncQueue() {
    if (syncInProgressRef.current) return
    syncInProgressRef.current = true

    try {
      const pendingItems = await getPendingItems()

      for (const item of pendingItems) {
        try {
          const client = await getAuthenticatedClient()
          const payload = JSON.parse(item.payload)

          // Sync based on type
          if (item.type === 'meal') {
            await client.post('/api/nutrition/meals', payload)
          } else if (item.type === 'photo') {
            // Upload photo with multipart form data
            await client.post('/api/nutrition/photos', payload)
          } else if (item.type === 'goal') {
            await client.put('/api/nutrition/goals', payload)
          }

          // Mark as synced
          await markSynced(item.id)
          console.log(`Synced: ${item.type} (${item.id})`)
        } catch (error) {
          console.error(`Sync failed for ${item.id}:`, error)

          // Get backoff delay and schedule retry
          const delayMs = await retryWithBackoff(item.id)

          // Clear existing timeout if any
          const existingTimeout = retryTimeoutsRef.current.get(item.id)
          if (existingTimeout) {
            clearTimeout(existingTimeout)
          }

          // Schedule retry after exponential backoff
          const timeout = setTimeout(() => {
            flushSyncQueue()
          }, delayMs)

          retryTimeoutsRef.current.set(item.id, timeout)
        }
      }
    } catch (error) {
      console.error('Sync queue error:', error)
    } finally {
      syncInProgressRef.current = false
    }
  }

  // Cleanup timeouts on unmount
  useEffect(() => {
    return () => {
      retryTimeoutsRef.current.forEach((timeout) => clearTimeout(timeout))
    }
  }, [])

  return { flushSyncQueue, isOnline }
}
```
